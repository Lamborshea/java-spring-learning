L08 配置管理

⏱ 1.5 小时 📦 Spring Boot 核心 🎯 application.yml

本节目标

  • 掌握 application.yml 语法
  • 理解多环境配置（dev / prod）
  • 会用 @Value 和 @ConfigurationProperties

1. 配置文件

Spring Boot 用 application.yml（或 .properties）做配置。YAML 比 properties 更可读。

# application.yml
server:
  port: 8080
  servlet:
    context-path: /api   # 所有接口前缀

spring:
  application:
    name: my-app
  datasource:
    url: jdbc:mysql://localhost:3306/test
    username: root
    password: 123456

app:
  name: My Spring Boot App
  version: 1.0.0
  feature:
    cache: true
    debug: false

2. @Value 单值注入

@Service
public class AppInfo {

  @Value("${app.name}")
  private String appName;

  @Value("${app.version:1.0.0}")  // 默认值
  private String version;

  @Value("${app.feature.cache}")
  private boolean cacheEnabled;
}

3. @ConfigurationProperties 批量注入

@Configuration
@ConfigurationProperties(prefix = "app")
@Data  // Lombok 自动生成 getter/setter
public class AppProperties {
  private String name;
  private String version;
  private Map<String, Boolean> feature;
}

或者用 @ConfigurationProperties + 构造方法（推荐）：

@ConfigurationProperties(prefix = "app")
public record AppProperties(
    String name,
    String version,
    Map<String, Boolean> feature
) {}

// 使用
@Service
@RequiredArgsConstructor
public class AppInfo {
  private final AppProperties props;

  public void print() {
    System.out.println(props.name());  // "My Spring Boot App"
  }
}

4. 多环境配置

src/main/resources/
├── application.yml          # 默认配置（所有环境共享）
├── application-dev.yml      # 开发环境
├── application-test.yml     # 测试环境
└── application-prod.yml     # 生产环境
# application-dev.yml
server:
  port: 8080
spring:
  datasource:
    url: jdbc:h2:mem:devdb
logging:
  level:
    com.learning: DEBUG

---
# application-prod.yml
server:
  port: 80
spring:
  datasource:
    url: jdbc:mysql://prod-host:3306/proddb
logging:
  level:
    com.learning: INFO

激活方式：

# 命令行
java -jar app.jar --spring.profiles.active=prod

# 环境变量
SPRING_PROFILES_ACTIVE=prod java -jar app.jar

# IDE：在 Run Configuration → Active profiles 填 dev

5. 配置加载顺序（优先级从高到低）

  1. 命令行参数 --server.port=9090
  2. 环境变量 SERVER_PORT=9090
  3. application-{profile}.yml
  4. application.yml
  5. 代码默认值
生产建议
敏感信息（密码、密钥）放环境变量或 application-prod.yml（不入 Git），而不是硬编码。
